    States — geocatalog.states¶

    Query and load model states (analyses, forecasts, ensemble members) from a catalog built with build_state_catalog. The how-to is Model states.

    Loading (extras: [xarray-raster])¶

    geocatalog.states.load_states(catalog, slice_, *, data_vars=None, group_by=None, storage_options=None) ¶

    Load forecast states as an (init_time, member, lead_time, y, x) cube.

    Every row overlapping slice_ is one run and member. Each file is clipped to the slice (bounds and valid-time interval), its valid times are relabelled as leads, τ = v - r, and the files are stacked by run time and member. Two runs valid at the same time stay two states, unlike load_xarray, which keeps one valid time. A member or lead missing from a run comes back as NaN, not as a shorter cube, and valid_time = init_time + lead_time is a 2-D coordinate. Members are in natural order: by number when every label is an integer ("2" before "10"), else as text (r1i1p1f1). A run whose file overlaps the slice without a time step inside it has no state there and is left out.

    def load_states(
    catalog: GeoCatalog,
    slice_: GeoSlice,
    *,
    data_vars: Sequence[str] | None = None,
    group_by: str | None = None,
    storage_options: dict[str, Any] | None = None,
) -> xr.Dataset | xr.DataTree:
    """Load forecast states as an ``(init_time, member, lead_time, y, x)`` cube.

    Every row overlapping ``slice_`` is one run and member. Each file is
    clipped to the slice (bounds and valid-time interval), its valid
    times are relabelled as leads, ``τ = v - r``, and the files are
    stacked by run time and member. Two runs valid at the same time stay
    two states, unlike `load_xarray`, which keeps one valid time. A
    member or lead missing from a run comes back as NaN, not as a shorter
    cube, and ``valid_time = init_time + lead_time`` is a 2-D coordinate.
    Members are in natural order: by number when every label is an
    integer (``"2"`` before ``"10"``), else as text (``r1i1p1f1``). A run
    whose file overlaps the slice without a time step inside it has no
    state there and is left out.

    Args:
        catalog: A state catalog (`build_state_catalog`), on either
            backend; filter it first with the typed queries
            (`forecasts`, `ensemble_at`, …).
        slice_: The region and valid-time interval to load. Bounds may be
            in another CRS; they are reprojected to the catalog's.
        data_vars: Data variables to keep. ``None`` keeps them all.
        group_by: A row column (``"producer"``) to split the rows by.
            ``None`` returns one cube and requires one grid; a column
            returns an `xarray.DataTree` with one leaf per value, each
            leaf a cube on its own grid.
        storage_options: obstore options for remote files (`geocloud.fs`).

    Returns:
        An `xarray.Dataset` with dims ``(init_time, member, lead_time,
        y, x)`` (the files' own spatial names), or an `xarray.DataTree`
        of them when ``group_by`` is given. A deterministic run (null
        member) is the member ``""``.

    Raises:
        ImportError: If the ``[xarray-raster]`` extra is missing.
        TypeError: If the catalog is not of kind ``"xarray"``.
        ValueError: If the catalog has no state columns, no row matches,
            the slice holds none of the matching files' time steps, a row
            has no run time, a run and member appears twice, or the rows
            are on different grids without ``group_by``.

    Example:
        >>> from geocatalog.states import forecasts, load_states
        >>> cube = load_states(forecasts(catalog), aoi)
        >>> cube["xch4"].dims, cube["xch4"].shape
        (('init_time', 'member', 'lead_time', 'lat', 'lon'), (3, 4, 2, 4, 5))
    """
    if xr is None:
        raise missing_extra("`load_states`", "xarray-raster")
    if catalog.kind != "xarray":
        raise TypeError(
            f"load_states requires an xarray-kind catalog; got {catalog.kind!r}"
        )
    require_state_columns(catalog, "load_states")
    filtered = catalog.query(slice_)
    rows = list(filtered.iter_rows())
    if not rows:
        raise ValueError("load_states: no catalog rows match the slice")
    bounds = _reproject_bounds(slice_.bounds, slice_.crs, filtered.crs)
    interval = (slice_.interval.left, slice_.interval.right)

    def read(row: CatalogRow) -> xr.Dataset:
        return _read_clipped(
            row.filepath,
            bounds=bounds,
            interval=interval,
            time_var=str(row.extras.get("time_var") or "time"),
            data_vars=data_vars,
            storage_options=storage_options,
        )

    # Catalog order is arbitrary (a SQL relation has none): sort first.
    rows.sort(key=lambda r: r.filepath)
    if group_by is None:
        return _stack(rows, read)
    if group_by not in rows[0].extras:
        raise ValueError(f"load_states: group_by column {group_by!r} not in catalog")
    groups: dict[str, list[CatalogRow]] = {}
    for row in rows:
        value = state_text(row.extras.get(group_by))
        if value is None or "/" in value:
            raise ValueError(
                f"load_states: {row.filepath} has {group_by}={value!r}; a group "
                "needs a non-null name without '/'."
            )
        groups.setdefault(value, []).append(row)
    return xr.DataTree.from_dict(
        {f"/{name}": _stack(group, read) for name, group in sorted(groups.items())}
    )

    def lead_window(
    catalog: GeoCatalog, start: pd.Timedelta, end: pd.Timedelta
) -> GeoCatalog:
    """Forecast rows whose leads overlap ``[start, end]``.

    A row covers valid times ``[v_min, v_max]``, so its leads are
    ``[v_min - run_time, v_max - run_time]``; lead time is derived, not
    stored.

    Args:
        catalog: A state catalog.
        start: First lead of the window, a `pd.Timedelta`.
        end: Last lead of the window (inclusive).

    Returns:
        A catalog of the forecast rows with at least one lead in the
        window.

    Raises:
        TypeError: If a bound is not timedelta-like.
        ValueError: If ``start > end``.

    Example:
        >>> day_two = lead_window(catalog, pd.Timedelta("2D"), pd.Timedelta("2D"))
    """
    lo, hi = _timedelta(start, "start"), _timedelta(end, "end")
    if lo > hi:
        raise ValueError(f"lead_window: start {lo} is after end {hi}")
    return _filter(
        catalog,
        [("kind", "==", "forecast"), ("lead_end", ">=", lo), ("lead_start", "<=", hi)],
        "lead_window",
    )

    def latest_analysis(catalog: GeoCatalog, *, before: pd.Timestamp) -> CatalogRow | None:
    """The analysis with the latest valid time at or before ``before``.

    Filters on the reserved ``kind`` column, not on a zero lead: a lead-0
    forecast row is valid at its run time too. Analyses valid at the same
    time are ordered by run time, then by file path (the last wins), so
    every backend returns the same row.

    Args:
        catalog: A state catalog.
        before: The cut-off valid time, a `pd.Timestamp` (inclusive).

    Returns:
        The `CatalogRow` of that analysis, or ``None`` when there is none.

    Example:
        >>> row = latest_analysis(catalog, before=pd.Timestamp("2026-01-03 06:00"))
        >>> row.filepath
        'ana_20260103.nc'
    """
    ts = _timestamp(before, "before")
    rows = _filter(
        catalog,
        [("kind", "==", "analysis"), ("end_time", "<=", ts)],
        "latest_analysis",
    )
    best: CatalogRow | None = None
    for row in rows.iter_rows():
        if best is None or _recency(row) > _recency(best):
            best = row
    return best

    Columns and name maps¶

    geocatalog.states.STATE_COLUMNS = ('kind', 'run_time', 'member', 'producer', 'parents') module-attribute ¶

    Row columns reserved, with fixed types, for model states (schema v1).

    kind (string, a StateKind), run_time (datetime64[us], naive UTC: the run's initialisation time), member (string: a CF realization number written as text, or a CMIP variant label such as r1i1p1f1), producer (string) and parents (list<string>: the URIs the state was computed from). A state catalog is one with a run_time column (is_state_frame), and it carries all five; in any other catalog the names are ordinary extras. Lead time is not stored: it is the row's valid time minus run_time. They stay ordinary row columns (and CatalogRow.extras keys), unlike RESERVED_COLUMNS, which every artifact carries.
